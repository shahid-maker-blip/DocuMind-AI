
cmake_minimum_required(VERSION 3.20)

project(DocuMindAI LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(PkgConfig REQUIRED)

pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SQLITE3 REQUIRED IMPORTED_TARGET sqlite3)

add_executable(DocuMind
    desktop/src/main.cpp

    desktop/imgui/imgui.cpp
    desktop/imgui/imgui_draw.cpp
    desktop/imgui/imgui_tables.cpp
    desktop/imgui/imgui_widgets.cpp

    desktop/imgui/backends/imgui_impl_sdl2.cpp
    desktop/imgui/backends/imgui_impl_sdlrenderer2.cpp
)

target_include_directories(DocuMind PRIVATE
    desktop/imgui
    desktop/imgui/backends
)

target_link_libraries(DocuMind PRIVATE
    PkgConfig::SDL2
    PkgConfig::SQLITE3
)

if(WIN32)
    target_link_libraries(DocuMind PRIVATE
        user32
        opengl32
        comdlg32
        shell32
        winhttp
    )
endif()
